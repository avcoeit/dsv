# PRACTICAL 01 - DATA MANIPULATION USING PANDAS

import pandas as pd

# Creating a DataFrame from a list
data = [['Ram', 18], ['Shyam', 16], ['Shiv', 20]]
df = pd.DataFrame(data, columns=['Name', 'Age'])
print(df)

# Adding a new column
data = {'Name': ['Ram', 'Shyam', 'Shiv'], 'Age': [18, 16, 20]}
df = pd.DataFrame(data)
address = ['New Delhi', 'Mumbai', 'Kolkata']
df['Address'] = address
print(df)

# Adding a new row
data = {'Name': ['Ram', 'Shyam', 'Shiv'],
        'Age': [18, 16, 20],
        'Address': ['New Delhi', 'Mumbai', 'Kolkata']}
df = pd.DataFrame(data)
print("Original DataFrame:")
print(df)
print()

df.loc[len(df.index)] = ['Ganesh', 12, 'Hyderabad']
print("Modified DataFrame:")
print(df)

# Deleting rows
data = {'Name': ['Ram', 'Shyam', 'Shiv', 'Radha', 'Gauri', 'Girija'],
        'Age': [25, 20, 30, 18, 26, 25],
        'City': ['Delhi', 'Mumbai', 'Kailash', 'Ujjain', 'Chennai', 'Shimla']}
df = pd.DataFrame(data)
print("Original DataFrame:")
print(df)
print()

df.drop(4, axis=0, inplace=True)
df.drop(index=5, inplace=True)
df.drop([1, 3], axis=0, inplace=True)
print("Modified DataFrame:")
print(df)

# Deleting columns
data = {'Name': ['Ravi', 'Anil', 'Mukseh', 'Rahul'],
        'Age': [25, 30, 35, 40],
        'City': ['New Delhi', 'Bhopal', 'Varanasi', 'Pune'],
        'Height': ['165', '178', '185', '171'],
        'Profession': ['Engineer', 'Entrepreneur', 'Unemployed', 'Actor'],
        'Marital Status': ['Single', 'Married', 'Divorced', 'Engaged']}
df = pd.DataFrame(data)
print("Original DataFrame:")
print(df)
print()

df.drop('Age', axis=1, inplace=True)
df.drop(columns='Marital Status', inplace=True)
df.drop(['Height', 'Profession'], axis=1, inplace=True)
print("Modified DataFrame:")
print(df)

# Renaming columns
data = {'Name': ['Alice', 'Bob', 'Charlie', 'David'],
        'Age': [25, 30, 35, 40],
        'City': ['New York', 'London', 'Paris', 'Tokyo']}
df = pd.DataFrame(data)
print("Original DataFrame:")
print(df)
print()

df.rename(columns={'Name': 'First_Name'}, inplace=True)
df.rename(mapper={'Age': 'Number', 'City': 'Address'}, axis=1, inplace=True)
print("Modified DataFrame:")
print(df)

# Renaming index labels
data = {'Name': ['Alice', 'Bob', 'Charlie', 'David'],
        'Age': [25, 30, 35, 40],
        'City': ['New York', 'London', 'Paris', 'Tokyo']}
df = pd.DataFrame(data)
print("Original DataFrame:")
print(df)
print()

df.rename(index={0: 7}, inplace=True)
df.rename(mapper={1: 10, 2: 100}, axis=0, inplace=True)
print("Modified DataFrame:")
print(df)
